# Governor Bridge: extra AI budget while a governor directive is active.
#
# The harness sets one country flag, governor_directive_<name>, on the player's empire (which the
# game's AI plays under human_ai). Each entry below applies only while its flag is set. Entries are
# additive (the ai_budget README: "Several entries may share the same category and resource"), so no
# vanilla file is overridden. Weights are relative to all other entries for the same resource
# (vanilla: alloys for ships 0.6, alloys for expansion starbases 0.2).

# --- expand: outposts and colonies ------------------------------------------------------------
governor_expand_alloys_starbases = {
	resource = alloys
	type = expenditure
	category = starbases
	potential = { has_country_flag = governor_directive_expand }
	weight = { weight = 0.3 }
	desired_min = { base = 100 }
}
# Outposts draw influence from `stations` (vanilla influence_expenditure_stations: "min is 1 jump
# away"); influence in `starbases` is only spent by nomadic empires, so it must not go there.
# Vanilla also only funds outpost alloys while influence > 75, so influence is the expansion gate.
governor_expand_influence_stations = {
	resource = influence
	type = expenditure
	category = stations
	potential = { has_country_flag = governor_directive_expand }
	weight = { weight = 0.3 }
	desired_min = { base = 75 }
}
governor_expand_alloys_colonies = {
	resource = alloys
	type = expenditure
	category = colonies
	potential = { has_country_flag = governor_directive_expand }
	weight = { weight = 0.2 }
}
